(defun +custom/get-file-path ()
  "Retrieve the file path of the current buffer.
Returns:
  - A string containing the file path in case of success.
  - `nil' in case the current buffer does not have a directory."
  (when-let (file-path (buffer-file-name))
    (file-truename file-path)))

(defun +custom/yank-filename ()
  "Copy and show the file name of the current buffer."
  (interactive)
  (if-let* ((file-path (+custom/get-file-path))
            (filename (file-name-nondirectory file-path)))
      (progn
        (kill-new filename)
        (message "Copied filename to clipboard: %s" filename))
    (message "Current buffer is not attached to a file!")))

(defun +custom/yank-filename-base ()
  "Copy and show the file name without its final extension of the current
buffer."
  (interactive)
  (if-let* ((file-path (+custom/get-file-path))
            (filename (file-name-base file-path)))
      (progn
        (kill-new filename)
        (message "Copied base filename to clipboard: %s" filename))
    (message "Current buffer is not attached to a file!")))

(defun +custom/yank-buffer-name ()
  "Copy and show the name of the current buffer."
  (interactive)
  (let ((buffer-name (buffer-name)))
    (kill-new buffer-name)
    (message "Copied buffer name to clipboard: %s" buffer-name)))

(defun +custom/yank-django-test-path ()
  "Copy the file path of the current buffer, replacing path separators with dots.
Removes any leading dots and file extension for Django test path format."
  (interactive)
  (if-let* ((file-path (+custom/get-file-path))
            (relative-path (file-relative-name file-path (projectile-project-root)))
            (without-extension (file-name-sans-extension relative-path))
            (dot-path (replace-regexp-in-string "/" "." without-extension))
            (cleaned-path (replace-regexp-in-string "^\\.+" "" dot-path)))
      (progn
        (kill-new cleaned-path)
        (message "Copied Django test path to clipboard: %s" cleaned-path))
    (message "Current buffer is not attached to a file!")))

(defun my/evil-find-file-at-point-with-line-other-window ()
  "Wrapper for evil-find-file-at-point-with-line that opens in another window."
  (interactive)
  (let ((file (ffap-file-at-point)))
    (if file
        (let ((new-window (or (split-window-sensibly)
                              (split-window))))  ; Fallback if split-window-sensibly returns nil
          (when new-window
            (select-window new-window)
            (evil-find-file-at-point-with-line)))
      (user-error "No file found at point"))))

(defun my/evil-find-file-at-point-with-line-right-window ()
  "Open file at point in the rightmost window, creating one if necessary."
  (interactive)
  (let ((file (ffap-file-at-point)))
    (if file
        (let ((right-window (window-in-direction 'right)))
          (if right-window
              (select-window right-window)
            (let ((new-window (split-window-right)))
              (when new-window
                (select-window new-window))))
          (evil-find-file-at-point-with-line))
      (user-error "No file found at point"))))

(defun check-process-running-p (process-name)
  "Check if the specified PROCESS-NAME is running.
Returns `t` if the process is running, and `nil` otherwise."
  (ignore-errors
    (let ((output (string-trim-right (shell-command-to-string (format "ps aux | grep %s | grep -v grep" process-name)))))
      (not (string-empty-p output)))))

(defun repeat-macro-until-end-of-buffer (macro-name)
  (interactive "SName of the macro: ")
  (condition-case nil
      (save-excursion
        (while t
          (call-interactively macro-name)
          (forward-line)
          (unless (looking-at-p "^[[:space:]]*$")
            (back-to-indentation))))

(defun reverse-text (start end)
  "Reverse characters between START and END."
  (interactive "r")
  (let ((text (buffer-substring start end)))
    (delete-region start end)
    (insert (concat (reverse (string-to-list text))))))

(defun get-string-in-parentheses ()
  "Returns the string within the nearest enclosing parentheses on the same line."
  (interactive)
  (save-excursion
    (let (content start end)
      ;; Move to the start of the current line to ensure searching from the beginning of the line
      (beginning-of-line)
      ;; Search forward to find an opening parenthesis
      (setq start (search-forward "(" (line-end-position) t))
      (when start
        (backward-char) ; Move back to the opening parenthesis
        (setq end (scan-sexps (point) 1)) ; Find the matching closing parenthesis
        (when end
          (goto-char start)
          (setq content (buffer-substring-no-properties start (- end 1))) ; Correctly extract the content without removing the first character
          content))))) ; Return the content

  "Returns a list of test names from lines starting with 'ERROR:' and extracts strings within parentheses."
  (interactive)
  (with-current-buffer (kizen-test--get-buffer)
    (save-excursion

          (let ((test-name (get-string-in-parentheses)))  ; Use the existing function

(defun get-test-names-from-doom-buffer ()
  "Returns a list of test names from Doom's scratch buffer.
Each line in the buffer is expected to be in the format:
'category / group / test.path.TestCase.test_name'
The function extracts the full test path (the part after the last '/')."
  (interactive)
  (with-current-buffer (doom-scratch-buffer)
    (save-excursion
      (goto-char (point-min))  ; Start from the beginning of the buffer
      (let (test-names)
        ;; Process each line in the buffer
        (while (not (eobp))  ; Continue until end of buffer
          (let* ((line (buffer-substring-no-properties
                        (line-beginning-position)
                        (line-end-position)))
                 (parts (split-string line " / "))
                 (test-name (car (last parts))))  ; Get the last element after splitting
            (when (and test-name (not (string-empty-p test-name)))
              (push (string-trim test-name) test-names)))
          (forward-line 1))
        ;; Return the list in the correct order (first test first)
        (nreverse test-names)))))

(defun kizen-test-compile-goto-error (&optional event)
  "Customized version of `compile-goto-error`."
  (interactive (list last-input-event))

  ;; Store the current point to return to in case of failure
  (let ((original-point (point))
        (line-contents (buffer-substring-no-properties (line-beginning-position) (line-end-position)))
        filename
        line-number)

    ;; Regexp to match filenames and line numbers in your compilation buffer
    (when (string-match "File \"\\([^\"]+\\)\", line \\([0-9]+\\)" line-contents)
      (setq filename (match-string 1 line-contents))
      (setq line-number (string-to-number (match-string 2 line-contents))))

    ;; Modify filename if it starts with "/app/apps"
    (when (and filename (string-match "^/app/apps" filename))
      (setq filename (replace-regexp-in-string "^/app/apps" "~/projects/kizen/webapp/apps" filename)))

    ;; Open the file in another window and go to the line number
    (if (and filename line-number
             (condition-case nil
                 (progn
                   (find-file-other-window filename)
                   (goto-char (point-min))
                   (forward-line (1- line-number))
                   (evil-scroll-line-to-center nil)  ; Scroll the line to the center
                   t)
               (error nil)))
        ;; Successfully navigated to the error
        (message "Navigated to custom error location.")
      ;; If anything fails, just call the original function and return to the original point
      (progn
        (goto-char original-point)
        (compile-goto-error event)))))

(defcustom copilot-chat-pr-prompt
  "Generate a focused pull request description using the following structure:

## Goal
Provide a concise summary of the main changes and their purpose. If there's a JIRA ticket, include its link.

Then, provide a summary of the most important changes, organized by category (like 'Core Changes', 'Bug Fixes', 'Improvements', etc.). For the most significant changes:
,* Use backticks to highlight important file paths
,* Focus on the key changes and their impact
,* Only include the most relevant implementation details
,* Highlight any breaking changes or important dependencies

Note: Focus only on the most important changes that other developers should know about, rather than listing every modified file.

Here are the changes to analyze:"
  "Prompt used for generating structured pull request descriptions."
  :type 'string
  :group 'copilot-chat)

(defun copilot-chat--get-pr-diff ()
  "Get the diff for PR using magit."
  (let ((head (magit-get-current-branch))
        (base (magit-get-push-branch)))
    (when (and base head)
      (with-temp-buffer
        (magit-git-insert "diff" base head)
        (buffer-string)))))

(defvar-local copilot-chat--accumulated-response nil
  "Accumulated response from Copilot for PR description.")

(defun copilot-chat-generate-pr-description ()
  "Generate a PR description in the current forge buffer."
  (interactive)
  (when (derived-mode-p 'forge-post-mode)
    (unless (copilot-chat--ready-p)
      (copilot-chat-reset))
    (let* ((diff (copilot-chat--get-pr-diff))
           (prompt (concat copilot-chat-pr-prompt diff))
           (inhibit-read-only t)
           (description-start nil)
           (current-buffer (current-buffer)))  ; Store the current buffer
      (if diff
          (save-excursion
            (goto-char (point-min))
            ;; Delete any existing content in the description area
            (when (re-search-forward "^title: " nil t)
              (forward-line 1)
              (setq description-start (point))
              (delete-region description-start (point-max)))
            ;; Reset accumulated response
            (setq-local copilot-chat--accumulated-response nil)
            ;; Create callback that accumulates the response
            (copilot-chat--ask
             prompt
             (lambda (content)
               (when (and (buffer-live-p current-buffer)
                         (not (string= content copilot-chat--magic)))
                 (with-current-buffer current-buffer
                   (setq-local copilot-chat--accumulated-response
                             (concat copilot-chat--accumulated-response content))))
               ;; When we get the magic string, insert the accumulated response
               (when (string= content copilot-chat--magic)
                 (with-current-buffer current-buffer
                   (let ((inhibit-read-only t))
                     (save-excursion
                       (goto-char description-start)
                       (insert "\n" copilot-chat--accumulated-response))))))
             t))
            (message "Generating PR description..."))
        (message "No changes detected between current branch and its push branch."))))

;; Bind to a key in forge-post-mode-map
(with-eval-after-load 'forge
  (define-key forge-post-mode-map (kbd "C-c C-g") #'copilot-chat-generate-pr-description))

  "Generate a PEP 257 compliant Python docstring for the current function."
  (interactive)
  (unless (copilot-chat--ready-p)
    (copilot-chat-reset))
  (save-excursion
    (mark-defun)
    (let* ((code (buffer-substring-no-properties (region-beginning) (region-end)))
           (prompt "Generate a PEP 257 compliant Python docstring for this code. The docstring should start and end with triple double-quotes, include Args/Returns/Raises sections as needed, and follow the standard format, only return the docstring directly and start your response with a newline, do not include the given function or class:")

      (deactivate-mark)
      (copilot-chat--insert-and-send-prompt formatted-prompt))))
