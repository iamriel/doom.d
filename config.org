* Doom Core
** Performance / GC / scrolling
#+begin_src emacs-lisp
;; NOTE: Doom uses gcmh, which *dynamically manages* `gc-cons-threshold' and
;; holds it at `gcmh-high-cons-threshold' (default 16MB) during activity -- so
;; setting `gc-cons-threshold' directly here is a no-op.  The real lever is
;; gcmh's threshold.  16MB is far too low for streaming subprocess output
;; (e.g. a `--force-color' kizen-test run): allocations blow past it
;; repeatedly, triggering a GC storm that -- with each GC forcing a redisplay
;; of the long colored lines -- freezes Emacs near the end of a run.
(setq! gc-cons-threshold (* 100 1024 1024)) ; 100MB (effective only pre-gcmh)
(after! gcmh
  (setq gcmh-high-cons-threshold (* 256 1024 1024))) ; 256MB during activity
;; Read process output in larger chunks so high-volume subprocess output is
;; handled in fewer passes -- fewer comint filter/scroll/redisplay cycles.
(setq! read-process-output-max (* 4 1024 1024)) ; 4MB
(setq! scroll-conservatively 101
       create-lockfiles nil)
#+end_src

** Strip assignment target
Turn =response = self.api.foo(x, is_async=False)= into =self.api.foo(x, is_async=False)=
from anywhere on the line. Only a plain top-level =​=​= is removed, so keyword args
(=x=y=), comparisons (===​==), and augmented/walrus (=+=​=/=:=​=) are left alone.
#+begin_src emacs-lisp
(defun +custom/remove-assignment ()
  "Strip the `target =' assignment prefix on the current line.
Keeps indentation and the right-hand side; works from anywhere on the line."
  (interactive)
  (save-excursion
    (beginning-of-line)
    (skip-chars-forward " \t")
    (let ((start (point))
          (eol   (line-end-position))
          (depth 0)
          (eqend nil))
      (save-excursion
        (goto-char start)
        (while (and (not eqend) (< (point) eol))
          (let ((c (char-after)))
            (cond
             ((memq c '(?\( ?\[ ?\{)) (setq depth (1+ depth)))
             ((memq c '(?\) ?\] ?\})) (setq depth (1- depth)))
             ((and (= depth 0) (eq c ?=)
                   (not (eq (char-after (1+ (point))) ?=))
                   (not (memq (char-before (point))
                              '(?= ?! ?< ?> ?: ?+ ?- ?* ?/ ?% ?| ?& ?^ ?@ ?~))))
              (setq eqend (1+ (point))))))
          (forward-char 1)))
      (if eqend
          (progn
            (goto-char eqend)
            (skip-chars-forward " \t")
            (delete-region start (point))
            (message "Stripped assignment target"))
        (message "No plain assignment `=' on this line")))))
#+end_src

** Avy: cross-pane copy
Grab text from another window without moving point. =avy-all-windows= lets the
copy commands reach into other panes; =+avy-copy-word= is a remote =yiw=.
#+begin_src emacs-lisp
(after! avy
  (setq avy-all-windows t))

(defun +avy-action-copy-word (pt)
  "Copy the inner word at PT to the kill ring (remote `yiw').
Restores the original window and point afterward."
  (save-excursion
    (goto-char pt)
    (when-let ((bounds (bounds-of-thing-at-point 'word)))
      (let ((str (buffer-substring (car bounds) (cdr bounds))))
        (kill-new str)
        (message "Copied: %s" str))))
  ;; jump back to where you were (avy stashed it in `avy-ring')
  (let ((dat (ring-ref avy-ring 0)))
    (select-frame-set-input-focus (window-frame (cdr dat)))
    (select-window (cdr dat))
    (goto-char (car dat)))
  t)

(defun +avy-copy-word ()
  "Remote `yiw': pick a word in any visible window, copy it to the kill ring.
Point and window stay put."
  (interactive)
  (require 'avy)
  (let ((avy-all-windows t))
    (avy-jump "\\b\\sw" :action #'+avy-action-copy-word)))

(defun +avy-insert-word ()
  "Pick a word from any visible window and INSERT it at point.
Made for insert state: you stay where you're typing, word lands at the cursor."
  (interactive)
  (require 'avy)
  (let ((origin     (point-marker))
        (origin-win (selected-window))
        (avy-all-windows t)
        str)
    (avy-jump
     "\\b\\sw"
     :action
     (lambda (pt)
       (save-excursion
         (goto-char pt)
         (when-let ((b (bounds-of-thing-at-point 'word)))
           (setq str (buffer-substring (car b) (cdr b)))))
       ;; back to where we were typing, then drop the word in
       (select-window origin-win)
       (goto-char origin)
       (when str (insert str))))
    (set-marker origin nil)))

;; Press `w' during ANY avy jump to copy that word.
(after! avy
  (setf (alist-get ?w avy-dispatch-alist) #'+avy-action-copy-word))

(after! which-key
  (setq which-key-idle-delay 0.5
        ;; NOTE: do not lower this toward 0.  Once a popup shows, which-key
        ;; restarts its idle timer as a *repeating* timer at this interval
        ;; (which-key.el `which-key--update' -> `which-key--start-timer').
        ;; At 0.01s it re-fires back-to-back while Emacs idles waiting on an
        ;; async process (e.g. a kizen-test run), rebuilding the popup ~100x/s
        ;; -- a redisplay+GC storm that freezes Emacs.  0.05 still feels
        ;; instant for chord typing but leaves idle headroom.
        which-key-idle-secondary-delay 0.05
        which-key-sort-order 'which-key-key-order-alpha))

  "Jump to the declaration of the Python class enclosing point."

;; Editor-only local overrides via a single gitignored `pyproject.local.toml'
;; marker at a project root (the "local pyproject"). Its PRESENCE is detected here
;; -- the file itself isn't parsed by this config, nor read by ruff/basedpyright/CI
;; -- and it flips two project-local behaviors, both buffer-local so the values are
;; live when `lsp!' starts the server (local-vars-hook fires after this mode hook):
;;   1. ruff LSP: opt into the broad ruleset as extendSelect (additive over the
;;      committed pyproject.toml's `select = ["B012"]', which is what CI's sheriff
;;      stage runs). Without the marker, ruff keeps the narrow committed set.
;;   2. basedpyright: drop to a rename/navigation-only server -- typeCheckingMode
;;      "off" (language services stay: rename, hover, definition, completion) plus

;;      Django apps, lambda runtimes) basedpyright can't statically resolve them, so
;;      reportMissingImports et al. would just flood the shared `lsp' flycheck
;;      checker on top of ruff. Without the marker, basedpyright type-checks normally.
;; CLI ruff, pre-commit, apheleia, and CI never see the marker (it's gitignored).
(add-hook! '(python-mode-hook python-ts-mode-hook)

           (locate-dominating-file buffer-file-name "pyproject.local.toml"))

(defun get-string-in-parentheses ()
  "Extract string within the nearest enclosing parentheses on the current line.
Returns nil if no parentheses are found or if they're malformed."
  (interactive)
  (save-excursion
    (beginning-of-line)
    (when-let* ((start (search-forward "(" (line-end-position) t))
                (end (progn (backward-char) (scan-sexps (point) 1))))
      (goto-char start)
      (buffer-substring-no-properties start (1- end)))))

(defun get-test-names-from-errors ()
  "Extract test names from lines starting with ERROR or FAIL in kizen-test buffer.
Returns a list of test names found within parentheses."
  (interactive)
  (require 'kizen-test)
  (when-let ((buffer (kizen-test--get-buffer)))
    (with-current-buffer buffer
      (save-excursion
        (goto-char (point-min))
        (let (test-names)
          (while (re-search-forward "^\\(ERROR\\|FAIL\\) .*(" nil t)
            (backward-char)
            (when-let ((test-name (get-string-in-parentheses)))
              (push test-name test-names)))
          (nreverse test-names))))))

(defun get-test-names-from-doom-buffer ()
  "Extract test names from Doom scratch buffer.
Each line should be in format: 'category / group / test.path.TestCase.test_name'
Returns the test path (part after the last '/')."
  (interactive)
  (when-let ((buffer (get-buffer "*doom:scratch*")))
    (with-current-buffer buffer
      (save-excursion
        (goto-char (point-min))
        (let (test-names)
          (while (not (eobp))
            (let* ((line (buffer-substring-no-properties
                          (line-beginning-position)
                          (line-end-position)))
                   (parts (split-string line " / "))
                   (test-name (car (last parts))))
              (when (and test-name (not (string-empty-p (string-trim test-name))))
                (push (string-trim test-name) test-names)))
            (forward-line 1))
          (nreverse test-names))))))

(defun kizen-test-compile-goto-error (&optional event)
  "Navigate to error location with Docker path mapping.
Maps /app/apps paths to ~/projects/kizen/webapp/apps for local development."
  (interactive (list last-input-event))
  (let ((original-point (point))
        (line-contents (buffer-substring-no-properties
                        (line-beginning-position)
                        (line-end-position))))

    ;; Try to extract filename and line number from current line
    (if-let* ((match (string-match "File \"\\([^\"]+\\)\", line \\([0-9]+\\)" line-contents))
              (filename (match-string 1 line-contents))
              (line-number (string-to-number (match-string 2 line-contents))))
        (progn
          ;; Map Docker paths to local paths
          (when (string-match "^/app/apps" filename)
            (setq filename (replace-regexp-in-string
                            "^/app/apps"
                            "~/projects/kizen/webapp/apps"
                            filename)))

          ;; Try to navigate to the file
          (condition-case err
              (progn
                (find-file-other-window filename)
                (goto-char (point-min))
                (forward-line (1- line-number))
                (when (fboundp 'evil-scroll-line-to-center)
                  (evil-scroll-line-to-center nil))
                (message "Navigated to %s:%d" filename line-number))
            (error
             (message "Failed to navigate to %s:%d - %s" filename line-number (error-message-string err))
             (goto-char original-point)
             (compile-goto-error event))))

      ;; Fall back to default behavior if no match found
      (compile-goto-error event))))

;; server automatically via lsp-mode's built-in ruff client (clients/lsp-ruff.el).
;; basedpyright is installed per-venv (`pip install basedpyright'); each project's
;; virtualenv is activated via pyvenv, which prepends its bin/ to `exec-path', so

;; basedpyright "wasn't working" before. lsp-pyright builds the `pyright' dependency
;; :system path as (concat lsp-pyright-langserver-command "-langserver") at LOAD
;; time, which is *before* this `after!' block sets the variable, so out of the box
;; it looks for a nonexistent "pyright-langserver" and the server never starts.
;; Re-register it with the correct name so it resolves "basedpyright-langserver"
;; from the active venv's bin. The variable itself stays the literal "basedpyright"
;; because lsp-pyright also reuses it to build settings keys / notification prefixes
;; (basedpyright.typeCheckingMode, basedpyright/beginProgress, ...); an absolute
;; path or a different string there would mangle those.

  ;; `basedpyright.analysis.*' namespace (verified by driving the server directly:

;; Terminal-style UIs (agent-shell banners, vterm, box drawing, spinners)
;; only line up if every glyph fills whole cells of the default
;; font. Doom picks fallback fonts after this file loads and replaces any
;; `set-fontset-font' made here; on macOS that leaves most symbols to fonts
;; sized for prose: Apple Symbols (3-19px wide in a 9px cell, even for — … →
;; that JetBrains Mono has), Apple Braille (10px), Apple Color Emoji (20x25px,
;; also for one-cell symbols like ⏺ ✔), Symbols Nerd Font Mono (15px).
;; Rebuild the order once Doom is done: the default font for whatever it
;; covers, then JuliaMono (same 0.6em advance, wide symbol coverage), then
;; Noto Sans Symbols 2 for old pictographs nothing else has (🕀 🖀), then
;; per-script fonts scaled below to land on the grid.
;;
;; Emacs moves fonts matching the face's own font to the front of a
;; character's list, so the default font wins wherever it has a glyph no
;; matter the order. Two-cell characters (emoji, CJK) must not use its
;; one-cell glyphs (⚡, ｛), so their lists are replaced rather than prepended
;; to. Frame fontsets (fontset-auto*) carry the frame font for every
;; character too, so this runs on all fontsets, not just the default.
(defun +my/cell-aligned-fontset-h ()
  "Order fallback fonts so non-ASCII text keeps to the default font's grid."
  (let ((default (font-spec :family (face-attribute 'default :family)))
        (emoji (or doom-emoji-font
                   (cl-find-if (doom-rpartial #'member (font-family-list))
                               doom-emoji-fallback-font-families))))
    (dolist (fontset (fontset-list))
      ;; Without ADD this replaces every range (Doom's too), so the result
      ;; is the same however often the hook runs.
      (set-fontset-font fontset 'unicode (font-spec :family "JuliaMono"))
      (set-fontset-font fontset 'unicode (font-spec :family "Noto Sans Symbols 2")
                        nil 'append)
      (set-fontset-font fontset 'unicode default nil 'prepend)
      (dolist (script '(symbol mathematical))
        (set-fontset-font fontset script (font-spec :family "Apple Symbols") nil 'append))
      (when emoji
        (set-fontset-font fontset 'emoji emoji)
        (set-fontset-font fontset 'symbol emoji nil 'append))
      (dolist (script '(han kana cjk-misc))
        (set-fontset-font fontset script (font-spec :family "Hiragino Sans")))
      (set-fontset-font fontset 'hangul (font-spec :family "Apple SD Gothic Neo"))
      ;; Nerd Font icons: JuliaMono has unrelated glyphs at some of these.
      (dolist (range '((#xe000 . #xf8ff) (#xf0000 . #xfffff)))
        (set-fontset-font fontset range (font-spec :family "Symbols Nerd Font Mono"))
        (set-fontset-font fontset range default nil 'prepend)))))
(add-hook 'after-setting-font-hook #'+my/cell-aligned-fontset-h)

** perf/jit-lock (defer only the first parse, keep scrolling instantly colored)
Tree-sitter font-lock of a freshly opened buffer (~treesit-font-lock-fontify-region~,
wrapped by ~indent-bars--fontify~) is the dominant first-open cost (~40% in
profiling) — the first parse of the file runs synchronously during ~find-file~.

Per ~jit-lock.el~, deferral only happens when the shared ~jit-lock-defer-timer~
exists (created only when ~jit-lock-defer-time~ is non-nil), and ~jit-lock-function~
defers when that timer exists *and* (~defer-time~ ≠ 0 *or* input is pending):
- a positive value ⇒ *always* defer ⇒ newly scrolled-in text flashes uncolored;
- ~0~ ⇒ defer only while input is pending, else fontify *synchronously when idle*
  ⇒ instant colors on scroll/buffer-switch, but the initial open (no input
  pending) would fontify synchronously and stall.

So: keep the global at ~0~ for instant colors everywhere, and force *always-defer*
buffer-locally for just the first moments of a freshly opened file — long enough
to take that one parse off the ~find-file~ path — then restore. Pairs with the
nav-flash change above (nothing else forces synchronous fontification on open).
#+begin_src emacs-lisp
(setq jit-lock-defer-time 0)

(defun +my/defer-initial-fontification-h ()
  "Defer only the first, parse-heavy fontification off the `find-file' path.
With `jit-lock-defer-time' globally 0, fontification is synchronous except while
input is pending — which does not cover the initial open — so force always-defer
for this buffer briefly, then restore so scrolling stays instantly colored."
  (setq-local jit-lock-defer-time 0.05)
  ;; Pass the buffer as a timer argument instead of closing over it: config.el is
  ;; tangled without lexical-binding, so a captured `buf' would be void when the
  ;; timer fires.
  (run-with-idle-timer
   0.3 nil
   (lambda (buf)
     (when (buffer-live-p buf)
       (with-current-buffer buf
         (kill-local-variable 'jit-lock-defer-time))))
   (current-buffer)))
(add-hook 'find-file-hook #'+my/defer-initial-fontification-h)
#+end_src

** ui/workspaces (reuse a project's workspace instead of duplicating it)
By default ~SPC p p~ runs ~+workspaces-switch-to-project-h~, which is *supposed*
to switch to an already-open project's workspace. It matches on the
~+workspace-project~ persp parameter, but that tag is only written when a
workspace is *created* for a project — not when an empty workspace is renamed
into one (the "non-empty" branch). Once a workspace is untagged, the lookup
falls through and a duplicate workspace is spawned every time you re-open the
project.

This replaces the switch action with one that finds the existing workspace
first (by stored project root, then by name), switches to it, and (re)tags it
so it always matches next time. Only when no workspace exists do we fall back
to Doom's default create-a-workspace behavior.
#+begin_src emacs-lisp
(after! persp-mode
  (defun +my/project-workspace-name (proot pname)
    "Return the name of the workspace already opened for project PROOT, or nil.
PNAME is the project's name. Matches first on the stored project root (robust to
renamed workspaces), then on a same-named workspace that isn't tagged to a
*different* project."
    (or
     (cl-loop for name in (+workspace-list-names)
              for root = (persp-parameter '+workspace-project (+workspace-get name t))
              when (and root (ignore-errors (file-equal-p root proot)))
              return name)
     (when-let* ((ws (+workspace-get pname t)))
       (let ((root (persp-parameter '+workspace-project ws)))
         (when (or (null root) (ignore-errors (file-equal-p root proot)))
           pname)))))

  (defun +my/switch-to-project-workspace (&optional dir)
    "Switch to the project's existing workspace, or create one for it.
Reliably reuses an already-open workspace for the same project instead of
spawning a duplicate the way `+workspaces-switch-to-project-h' can when a
workspace was never tagged with its project root."
    (let* ((default-directory (or dir default-directory))
           (proot (file-truename default-directory))
           (pname (doom-project-name))
           (existing (+my/project-workspace-name proot pname)))
      (if existing
          (progn
            (+workspace-switch existing)
            (+workspace-message
             (format "Switched to existing '%s' workspace" existing) 'success))
        (+workspaces-switch-to-project-h dir))
      ;; Always (re)tag the resulting workspace so future switches find it.
      (set-persp-parameter '+workspace-project proot (+workspace-current))))

  (setq projectile-switch-project-action #'+my/switch-to-project-workspace))

** Pre-warm heavy first-file init (Copilot server, Yasnippet)
Opening the first file of a session blocks Emacs while one-time init runs
synchronously on the ~find-file~ path. Profiling showed two leaves: Copilot's
~copilot--start-server~ makes a *synchronous* jsonrpc initialize that blocks on
~accept-process-output~ until the Node server is up (~20%), and Yasnippet's
first ~yas-minor-mode-on~ runs ~yas--load-pending-jits~ to load every snippet
table (~12%). Pay both during idle instead, so the live Copilot connection and
loaded snippets are ready before the first real open.
#+begin_src emacs-lisp
(defun +my/prewarm-heavy-packages ()
  "Pre-pay one-time init costs (Copilot, Yasnippet, LSP clients) during idle.
These otherwise block synchronously on the first `find-file' of the session."
  ;; Copilot: bring the Node server up now via its blocking initialize, so the
  ;; first prog-mode buffer reuses the live connection instead of stalling
  ;; `find-file'. `copilot--ensure-doc-open' then short-circuits on alive.
  (ignore-errors
    (require 'copilot)
    (unless (copilot--connection-alivep)
      (copilot--start-server)))
  ;; Yasnippet: snippet tables load lazily per major-mode on the first buffer of
  ;; that mode, stalling its first open (`yas--load-pending-jits' -> per-snippet
  ;; `set-auto-coding'). Force the tables (and tree-sitter grammars) for the
  ;; modes we use to load now, in throwaway buffers, so real opens are instant.
  (ignore-errors
    (require 'yasnippet)
    (yas-reload-all)
    (dolist (mode '(emacs-lisp-mode python-ts-mode js-ts-mode
                    typescript-ts-mode web-mode))
      (ignore-errors
        (with-temp-buffer
          (delay-mode-hooks (funcall mode))
          (yas-minor-mode 1)))))
  ;; LSP: load every client package now. The first `lsp' call otherwise runs
  ;; `lsp--require-packages' on the `find-file' path, loading all clients there.
  (ignore-errors
    (require 'lsp-mode)
    (lsp--require-packages)))

(run-with-idle-timer 1 nil #'+my/prewarm-heavy-packages)
#+end_src

Claude and Codex run through their ACP adapters (=claude-agent-acp= and
=codex-acp=, from =npm i -g=) in ordinary Emacs buffers instead of a terminal.
They share =~/.claude= and =~/.codex= with the CLIs, so a tmux session can be
resumed here (~SPC j r~) and the other way around, just not both at once.

- =lisp/agent-git= names every shell =Agent @ repo:worktree ⎇ branch= and
  renames it when a turn ends on another branch. The mode line shows where
  each shell works, and any buffer switcher finds it by branch or ticket.
- ~SPC j j~ lists every shell in every project with status and session title;
  ~SPC j w~ lists every worktree of every known repo; ~SPC j W~ creates a new
  branch in a sibling worktree (=webapp_feature-KZN-…=) and starts a shell.
- ~SPC j K~ removes a worktree: closes its shells, buffers and workspace,
  deletes it, and deletes its branch when every commit is on =origin/develop=

  =AGENTS.md=, =RTK.md=) are symlinked from the main checkout into a worktree

  ones too) and resumes the chosen one in the same shell; =/resume text=

(defun +agent/close-worktree-workspace-h (root)
  "Close ROOT's workspace and forget ROOT as a project: its worktree is going.
For `agent-git-remove-worktree-functions', which runs while ROOT exists."
  (when (and (bound-and-true-p persp-mode) (fboundp '+my/project-workspace-name))
    (when-let* ((name (let ((default-directory root))
                        (+my/project-workspace-name (file-truename root) (doom-project-name)))))
      (condition-case err
          (+workspace/kill name)
        (error (message "Left workspace %s open: %s" name (error-message-string err))))))
  (when (featurep 'projectile)
    (dolist (project projectile-known-projects)
      (when (and (not (file-remote-p project)) (file-equal-p project root))
        (projectile-remove-known-project project)))))

  ;; Fold headers, permission buttons, links and forms bind RET in a
  ;; `keymap' text property. In GUI frames the key arrives as <return>, which
  ;; evil-collection binds for comint buffers, so it never falls back to RET
  ;; and runs `newline' on read-only text instead. This binding hands
  ;; <return> to the RET command at point; elsewhere it returns the binding
  ;; it hides (submit in insert state, newline in normal state).
  (defun +agent--button-ret (_)
    (let ((map (get-char-property (point) 'keymap))
          command)
      (if (and (keymapp map)
               (commandp (setq command (lookup-key map (kbd "RET")))))
          command
        (when-let* ((aux (evil-get-auxiliary-keymap
                          (keymap-parent agent-shell-mode-map) evil-state)))
          (lookup-key aux [return])))))
  (map! :map agent-shell-mode-map

  ;; header and ? menu: the viewport (n/p next/previous, r reply, f/b pages,
  ;; y and 1-9 replies, ? help), the diff (n/p hunks, y accept, RET open
  ;; file, q quit) and the ACP traffic log (n/p entries). They open in
  ;; normal state, whose keys come first (n search, p paste, r replace,
  ;; digits count), and evil-collection only adds g-prefixed copies (gr, g?).
  ;; Copy each map's own keys into its Evil bindings so they win. Keys a view
  ;; doesn't bind (j/k, gg, /, V, SPC, the g copies) stay Evil's. Its parent
  ;; map's keys aren't copied: `special-mode-map' would take SPC and g.
  (defun +agent--own-keys-in-evil (map &rest states)
    "Bind MAP's own keys, not its parent's, for MAP in Evil STATES."
    (let ((own (copy-keymap map)))
      (set-keymap-parent own nil)
      (map-keymap
       (lambda (event def)
         (unless (or (memq event '(remap menu-bar))
                     ;; Evil's per-state maps, stored in MAP as normal-state etc.
                     (and (symbolp event)
                          (string-suffix-p "-state" (symbol-name event))))
           (dolist (state states)
             (evil-define-key* state map (vector event) def))))
       own)))

    (+agent--own-keys-in-evil agent-shell-viewport-view-mode-map 'normal)
